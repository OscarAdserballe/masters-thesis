\begin{table}[t]
\centering
\caption{Experimental design.}
\label{tab:design}
\begin{tabular}{@{}ll@{}}
\toprule
Component & Specification \\
\midrule
Task & Flight recommendation (Qiu et al., 2026); 3 flights per menu \\
Attributes & Departure time, duration, number of stops, price (scaled to $[0,1]$) \\
Principals & 50 simulated users, $\theta_i \in \{-1,-\tfrac12,0,\tfrac12,1\}^4$, stratified by $\lVert\theta_i\rVert$ \\
Interaction & 5 rounds; after each pick: ``correct'' or ``incorrect, I prefer Flight $k$'' \\
Evaluation & 100 held-out menus at 6 checkpoints (after 0--5 rounds), no feedback \\
Decisions & $50 \times 605 = 30{,}250$ per agent \\
Outcomes & Accuracy; utility regret; share of achievable welfare \\
Inference & 95\% CIs from resampling users (10{,}000 draws) \\
\bottomrule
\end{tabular}
\end{table}
